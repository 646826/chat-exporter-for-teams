function event() {
  const listeners = new Set();
  return { addListener(fn) { listeners.add(fn); }, removeListener(fn) { listeners.delete(fn); }, emit(...args) { for (const fn of [...listeners]) fn(...args); } };
}
function pair(sender) {
  const messages = [];
  const a = {name:'teams-file-download-v1',onMessage:event(),onDisconnect:event()};
  const b = {name:a.name,sender,onMessage:event(),onDisconnect:event()};
  let disconnected = false;
  for (const [from,to] of [[a,b],[b,a]]) {
    from.postMessage = message => { if(disconnected) throw new Error('port closed'); messages.push(message); queueMicrotask(()=>to.onMessage.emit(message)); };
    from.disconnect = () => { if(disconnected)return;disconnected=true;queueMicrotask(()=>to.onDisconnect.emit()); };
  }
  return {a,b,messages};
}

  const api={runtime:{id:'test-extension'},permissions:{contains:async()=>permissions},storage:{session:{

  }}};
  const session = await access.createDownloadSession(api,42,'document-1');

}

test('authorized transport preserves bytes with bounded messages and pull backpressure',async()=>{
  const bytes=Uint8Array.from({length:180000},(_,i)=>i%251);
  const env=await environment({fetchImpl:async()=>new Response(bytes,{headers:{'content-type':'application/pdf','content-disposition':'attachment; filename="report.pdf"'}})});
  const response=await client.fetchThroughExtension(url,{}, {...config,downloadSession:env.session},env.api.runtime);
  assert.equal(response.status,200);assert.equal(response.headers.get('content-type'),'application/pdf');
  assert.equal(env.ports[0].messages.filter(m=>m.type==='chunk').length,0,'no bytes before consumer read');
  assert.deepEqual(new Uint8Array(await response.arrayBuffer()),bytes);
  assert.ok(env.ports[0].messages.filter(m=>m.type==='chunk').every(m=>m.data.length<=65536));
  assert.equal(env.fetches,1);
});
